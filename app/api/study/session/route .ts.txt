import { NextResponse } from "next/server";
import {
  readUsers,
  writeUsers,
  userIdFromToken,
} from "@/app/api/_lib";

type StudySession = {
  userId: string;
  startedAt: number;
  endsAt: number;
  taskIndex: number;
  completed: boolean;
  completedAt?: number;
};

type StudyData = Record<string, StudySession>;

import fs from "node:fs/promises";
import path from "node:path";

const file = path.join(
  process.cwd(),
  "data",
  "study-sessions.json"
);

const tasks = [
  "راجع درسًا واحدًا بتركيز.",
  "اقرأ صفحة وحاول تلخيصها.",
  "ذاكر 10 كلمات جديدة.",
  "حل بعض مسائل الرياضيات.",
  "راجع أهم النقاط التي درستها اليوم.",
  "اقرأ موضوعًا مفيدًا واكتب 3 أفكار منه.",
  "راجع درسًا قديمًا بدون استخدام الهاتف.",
  "حل سؤالًا صعبًا وحاول تعرف طريقة الحل.",
];

async function readData(): Promise<StudyData> {
  try {
    return JSON.parse(
      await fs.readFile(file, "utf8")
    );
  } catch {
    return {};
  }
}

async function writeData(data: StudyData) {
  await fs.mkdir(path.dirname(file), {
    recursive: true,
  });

  await fs.writeFile(
    file,
    JSON.stringify(data, null, 2),
    "utf8"
  );
}

function getToken(req: Request) {
  const cookie = req.headers.get("cookie") || "";

  return cookie
    .split(";")
    .map((item) => item.trim())
    .find((item) =>
      item.startsWith("miftah_session=")
    )
    ?.split("=")
    .slice(1)
    .join("=");
}

function getUserId(req: Request) {
  return userIdFromToken(getToken(req));
}

export async function POST(req: Request) {
  try {
    const userId = getUserId(req);

    if (!userId) {
      return NextResponse.json(
        { error: "يجب تسجيل الدخول أولًا." },
        { status: 401 }
      );
    }

    const body = await req.json();
    const action = body?.action;

    const data = await readData();
    const existing = data[userId];

    if (action === "start") {
      if (existing && !existing.completed) {
        const remaining = Math.max(
          0,
          Math.ceil(
            (existing.endsAt - Date.now()) / 1000
          )
        );

        return NextResponse.json({
          ok: true,
          active: true,
          remaining,
          task: tasks[existing.taskIndex],
          taskIndex: existing.taskIndex,
        });
      }

      const nextIndex = existing
        ? (existing.taskIndex + 1) % tasks.length
        : 0;

      const startedAt = Date.now();

      const session: StudySession = {
        userId,
        startedAt,
        endsAt: startedAt + 15 * 60 * 1000,
        taskIndex: nextIndex,
        completed: false,
      };

      data[userId] = session;
      await writeData(data);

      return NextResponse.json({
        ok: true,
        active: true,
        remaining: 15 * 60,
        task: tasks[nextIndex],
        taskIndex: nextIndex,
      });
    }

    if (action === "complete") {
      const session = data[userId];

      if (!session) {
        return NextResponse.json(
          { error: "لا توجد جلسة مذاكرة." },
          { status: 400 }
        );
      }

      if (session.completed) {
        return NextResponse.json({
          ok: true,
          alreadyCompleted: true,
        });
      }

      if (Date.now() < session.endsAt) {
        const remaining = Math.ceil(
          (session.endsAt - Date.now()) / 1000
        );

        return NextResponse.json(
          {
            error: "لسه وقت المذاكرة مخلصش.",
            remaining,
          },
          { status: 400 }
        );
      }

      const users = await readUsers();
      const user = users.find(
        (item) => item.id === userId
      );

      if (!user) {
        return NextResponse.json(
          { error: "الحساب غير موجود." },
          { status: 404 }
        );
      }

      user.xp += 10;

      session.completed = true;
      session.completedAt = Date.now();

      data[userId] = session;

      await writeUsers(users);
      await writeData(data);

      return NextResponse.json({
        ok: true,
        added: 10,
        xp: user.xp,
        nextTask:
          tasks[
            (session.taskIndex + 1) % tasks.length
          ],
      });
    }

    return NextResponse.json(
      { error: "عملية غير صحيحة." },
      { status: 400 }
    );
  } catch (error) {
    console.error("STUDY SESSION ERROR:", error);

    return NextResponse.json(
      { error: "حدث خطأ في جلسة المذاكرة." },
      { status: 500 }
    );
  }
}
