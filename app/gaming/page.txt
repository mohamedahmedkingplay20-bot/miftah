"use client";

import Link from "next/link";
import { useEffect, useMemo, useState } from "react";

type Question = {
  question: string;
  options: string[];
  answer: string;
};

const QUESTIONS: Question[] = [
  {
    question: "أي لعبة تعتمد بشكل أساسي على عالم من المكعبات؟",
    options: ["Minecraft", "Fortnite", "FIFA", "Valorant"],
    answer: "Minecraft",
  },
  {
    question: "ما الشركة المطورة للعبة Fortnite؟",
    options: ["Epic Games", "Valve", "Mojang", "Ubisoft"],
    answer: "Epic Games",
  },
  {
    question: "على أي منصة توجد لعبة Blox Fruits؟",
    options: ["Roblox", "Steam", "PlayStation", "Xbox فقط"],
    answer: "Roblox",
  },
  {
    question: "أي لعبة تشتهر بشخصية Kratos؟",
    options: ["God of War", "Minecraft", "Fortnite", "Rocket League"],
    answer: "God of War",
  },
  {
    question: "أي لعبة تجمع بين السيارات وكرة القدم؟",
    options: ["Rocket League", "GTA V", "Valorant", "Among Us"],
    answer: "Rocket League",
  },
  {
    question: "أي لعبة تشتهر بنمط Battle Royale؟",
    options: ["Fortnite", "Minecraft Story Mode", "The Sims", "Portal"],
    answer: "Fortnite",
  },
];

function shuffle<T>(items: T[]) {
  return [...items].sort(() => Math.random() - 0.5);
}

export default function GamingPage() {
  const questions = useMemo(() => shuffle(QUESTIONS), []);

  const [current, setCurrent] = useState(0);
  const [score, setScore] = useState(0);
  const [selected, setSelected] = useState("");
  const [finished, setFinished] = useState(false);
  const [saving, setSaving] = useState(false);
  const [xpMessage, setXpMessage] = useState("");

  const question = questions[current];

  useEffect(() => {
    setSelected("");
  }, [current]);

  async function answer(option: string) {
    if (selected || finished) return;

    setSelected(option);

    const correct = option === question.answer;
    const newScore = correct ? score + 1 : score;

    if (correct) {
      setScore(newScore);
    }

    window.setTimeout(async () => {
      if (current >= questions.length - 1) {
        setFinished(true);

        const earnedXP = newScore * 5;

        if (earnedXP > 0) {
          setSaving(true);

          try {
            const response = await fetch("/api/xp/add", {
              method: "POST",
              headers: {
                "Content-Type": "application/json",
              },
              body: JSON.stringify({
                amount: earnedXP,
              }),
            });

            if (response.ok) {
              setXpMessage(`🔥 حصلت على ${earnedXP} XP`);
            } else {
              setXpMessage(
                "تم إنهاء التحدي، لكن سجل الدخول لحفظ XP."
              );
            }
          } catch {
            setXpMessage(
              "خلصت التحدي، لكن حصلت مشكلة في حفظ XP."
            );
          } finally {
            setSaving(false);
          }
        } else {
          setXpMessage("جرب مرة تانية وارفع نتيجتك 💪");
        }

        return;
      }

      setCurrent((value) => value + 1);
    }, 700);
  }

  function restart() {
    window.location.reload();
  }

  if (finished) {
    return (
      <main
        dir="rtl"
        className="min-h-screen bg-[#020617] px-5 py-10 text-white"
      >
        <div className="mx-auto flex min-h-[85vh] max-w-3xl items-center justify-center">
          <div className="w-full rounded-[35px] border border-white/10 bg-[#081120] p-10 text-center shadow-2xl">
            <div className="text-7xl">🏆</div>

            <h1 className="mt-6 text-5xl font-black">
              خلص التحدي!
            </h1>

            <p className="mt-4 text-lg text-slate-400">
              نتيجتك:
            </p>

            <div className="mt-3 text-6xl font-black text-blue-400">
              {score} / {questions.length}
            </div>

            <div className="mt-6 rounded-2xl border border-blue-500/20 bg-blue-500/10 p-5 font-black text-blue-300">
              {saving ? "جاري حفظ XP..." : xpMessage}
            </div>

            <div className="mt-8 flex flex-col gap-3 sm:flex-row">
              <button
                onClick={restart}
                className="flex-1 rounded-2xl bg-blue-600 px-6 py-4 font-black transition hover:bg-blue-500"
              >
                🎮 تحدي جديد
              </button>

              <Link
                href="/"
                className="flex-1 rounded-2xl border border-white/10 px-6 py-4 font-black transition hover:bg-white/5"
              >
                🏠 الرئيسية
              </Link>
            </div>
          </div>
        </div>
      </main>
    );
  }

  return (
    <main
      dir="rtl"
      className="min-h-screen bg-[#020617] px-5 py-10 text-white"
    >
      <div className="mx-auto max-w-4xl">
        <Link
          href="/"
          className="text-sm font-bold text-slate-400 transition hover:text-white"
        >
          → الرئيسية
        </Link>

        <div className="mt-10 text-center">
          <div className="text-6xl">🎮</div>

          <h1 className="mt-5 text-5xl font-black">
            تحدي الجيمرز
          </h1>

          <p className="mt-3 text-slate-400">
            السؤال بيتغير تلقائيًا بعد كل إجابة 🔥
          </p>
        </div>

        <div className="mt-10 rounded-[35px] border border-white/10 bg-[#081120] p-7 shadow-2xl md:p-10">
          <div className="mb-8 flex items-center justify-between">
            <span className="rounded-full bg-white/5 px-4 py-2 text-sm font-bold text-slate-400">
              السؤال {current + 1} من {questions.length}
            </span>

            <span className="rounded-full bg-blue-500/10 px-4 py-2 text-sm font-black text-blue-300">
              ⭐ {score} نقطة
            </span>
          </div>

          <h2 className="text-center text-3xl font-black leading-relaxed md:text-4xl">
            {question.question}
          </h2>

          <div className="mt-10 grid gap-4 md:grid-cols-2">
            {question.options.map((option) => {
              const isSelected = selected === option;
              const isCorrect = selected && option === question.answer;

              return (
                <button
                  key={option}
                  onClick={() => answer(option)}
                  disabled={Boolean(selected)}
                  className={`rounded-2xl border p-5 text-right font-black transition ${
                    isSelected
                      ? isCorrect
                        ? "border-emerald-400 bg-emerald-400/10 text-emerald-300"
                        : "border-red-400 bg-red-400/10 text-red-300"
                      : "border-white/10 bg-white/[0.03] hover:-translate-y-1 hover:border-blue-400/40 hover:bg-blue-500/5"
                  }`}
                >
                  {option}
                </button>
              );
            })}
          </div>

          <div className="mt-8 text-center text-sm text-slate-500">
            ✅ كل إجابة صحيحة = 5 XP
          </div>
        </div>
      </div>
    </main>
  );
}
